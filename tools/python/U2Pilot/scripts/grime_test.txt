# grime_test: U2Grime on Sanctuary. Dust at wall feet and corners away from the AI paths,
# worked out when the map loads. Needs U2Grime.u in System (EditPackages=U2Grime, UCC make);
# the mutator comes in through the map URL, nothing else to install.
# Results: "Grime:" lines in Unreal2.log (scan counts, spots placed, each view, wear steps)
# and screenshots: each of the 6 dustiest spots with dust, then without (same view); then
# the clutter (props copied from the map's own small ones), with and without, and a kick.
map M08A1?Mutator=U2Grime.GrimeMutator
waitcontrol 180
console god
# the analysis waits 1 s, then spreads over a few seconds
wait 8
dump GrimeManager Phase Navs Rays Walls MaxSpots
console set GrimeManager ViewSpot 0
wait 0.5
shot
console set GrimeManager bShow False
wait 0.3
shot
console set GrimeManager bShow True
console set GrimeManager ViewSpot 1
wait 0.5
shot
console set GrimeManager bShow False
wait 0.3
shot
console set GrimeManager bShow True
console set GrimeManager ViewSpot 2
wait 0.5
shot
console set GrimeManager bShow False
wait 0.3
shot
console set GrimeManager bShow True
console set GrimeManager ViewSpot 3
wait 0.5
shot
console set GrimeManager bShow False
wait 0.3
shot
console set GrimeManager bShow True
console set GrimeManager ViewSpot 10
wait 0.5
shot
console set GrimeManager bShow False
wait 0.3
shot
console set GrimeManager bShow True
console set GrimeManager ViewSpot 40
wait 0.5
shot
console set GrimeManager bShow False
wait 0.3
shot
console set GrimeManager bShow True
# wear: back in front of spot 0, walk onto it and stand-walk over it (strafe back and forth)
console set GrimeManager ViewSpot 0
wait 0.5
move 1 0 0.7
move 0 1 0.6
move 0 -1 0.6
move 0 1 0.6
move 0 -1 0.6
move 0 1 0.6
move 0 -1 0.6
move -1 0 0.7
wait 0.3
shot
# clutter: which props the map offered, then pieces with and without
dump GrimeClutter Phase Measured Unmeasured TooBig Floating
console set GrimeClutter ViewProp 0
wait 0.5
shot
console set GrimeClutter bShow False
wait 0.3
shot
console set GrimeClutter bShow True
console set GrimeClutter ViewProp 5
wait 0.5
shot
console set GrimeClutter bShow False
wait 0.3
shot
console set GrimeClutter bShow True
console set GrimeClutter ViewProp 15
wait 0.5
shot
# kick: walk into piece 0 (it should hop away and settle; "Kicks" counts it)
console set GrimeClutter ViewProp 0
wait 0.5
move 1 0 1.2
wait 1.5
shot
dump GrimeProp Kicks Physics
# break: stand in front of piece 5 and shoot it: it should burst into shards that bounce and lie
# there; BrokenByDamage = the engine's own hit reached it, BrokenByShot = the aim-trace fallback did
console set GrimeClutter ViewProp 5
wait 0.5
shot
fire 0.25
wait 0.4
shot
wait 1.5
shot
dump GrimeClutter Broken BrokenByDamage BrokenByShot
quit
